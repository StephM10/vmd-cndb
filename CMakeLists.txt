cmake_minimum_required(VERSION 3.24)
project(vmd_cndb VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ZLIB_USE_STATIC_LIBS ON)
find_package(ZLIB REQUIRED)
find_package(HDF5 CONFIG REQUIRED COMPONENTS C static)
add_library(cndb_core STATIC src/reader.cpp)
target_include_directories(cndb_core PUBLIC src)
target_link_libraries(cndb_core PUBLIC hdf5-static)
target_compile_definitions(cndb_core PUBLIC H5_USE_110_API)
if(MSVC)
  target_compile_options(cndb_core PRIVATE /W4 /utf-8 /permissive-)
endif()
add_executable(cndb_check src/check.cpp)
target_link_libraries(cndb_check PRIVATE cndb_core)

set(TCL_SOURCE_DIR "" CACHE PATH "Tcl 8.6 source tree (only the stub library is built)")
if(NOT EXISTS "${TCL_SOURCE_DIR}/generic/tclStubLib.c")
  message(FATAL_ERROR "Use scripts/build.ps1 or set TCL_SOURCE_DIR to Tcl 8.6 sources")
endif()
add_library(cndb_tclstub STATIC "${TCL_SOURCE_DIR}/generic/tclStubLib.c")
target_include_directories(cndb_tclstub PRIVATE "${TCL_SOURCE_DIR}/generic" "${TCL_SOURCE_DIR}/win" "${TCL_SOURCE_DIR}/unix")
target_compile_definitions(cndb_tclstub PRIVATE STATIC_BUILD USE_TCL_STUBS)
add_library(vmdcndb SHARED src/plugin.cpp)
target_include_directories(vmdcndb PRIVATE third_party/vmd third_party/tcl)
target_compile_definitions(vmdcndb PRIVATE VMDPLUGIN_EXPORTS USE_TCL_STUBS NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(vmdcndb PRIVATE cndb_core cndb_tclstub)
if(MSVC)
  target_compile_options(vmdcndb PRIVATE /W4 /utf-8 /permissive-)
endif()
set_target_properties(vmdcndb PROPERTIES PREFIX "" SUFFIX ".dll")
install(TARGETS vmdcndb cndb_check RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
install(DIRECTORY tcl/ DESTINATION tcl)
install(FILES scripts/install.ps1 scripts/uninstall.ps1 DESTINATION .)
install(FILES README.md LICENSE CHANGELOG.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY examples/ DESTINATION examples)
install(DIRECTORY third_party/licenses/ DESTINATION licenses)
install(FILES third_party/tcl/license.terms DESTINATION licenses RENAME Tcl.txt)
